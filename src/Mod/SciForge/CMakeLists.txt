# SciForge module: the Fusion-style interface layer and the golden-model suite.
# Pure Python for now, so this only copies files into the build and install trees.

# VERSION: SciForge's own version ("0.1.0-beta"), the one file the programs (src/Main), the
# installer, the release scripts and sciforge.__version__ all read.
SET(SciForge_Init
    Init.py
    InitGui.py
    VERSION
)

# Every module, icon and test file in the folder (never a list kept by hand: in October
# 2026 the list still had 35 of 116 modules, the full build missed the other 81 and
# failed with "cannot import name 'blend' from 'sciforge'"; the quick update copies the
# whole folder, so it never showed there). tools/sciforge/check_install.sh checks this
# on every push. CONFIGURE_DEPENDS: a new file is picked up without re-running CMake by
# hand. Recursive: sub-packages such as sciforge/ai/ops/ are included.
FILE(GLOB_RECURSE SciForge_Package RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    sciforge/*.py)
FILE(GLOB SciForge_Icons RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    sciforge/icons/*.svg)

# Ribbon/menu icons, generated by tools/sciforge/make_icons.py.
FILE(GLOB SciForge_UiIcons RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    sciforge/icons/ui/*.svg)

# Unit tests, golden models (their builder and the model files) and the GUI tests.
FILE(GLOB_RECURSE SciForge_Tests RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    SciForgeTests/*.py SciForgeTests/*.json SciForgeTests/*.md SciForgeTests/*.txt)

# Sample files the Insert/Export tests open (meshes, STEP, SVG, DXF, images): every file.
# (files only: the VEX test parts' generator sits in a sub-folder, installed with the tests)
FILE(GLOB SciForge_Fixtures LIST_DIRECTORIES false RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
    CONFIGURE_DEPENDS SciForgeTests/fixtures/*)

# What the Home tab's welcome line opens offline: the cheat sheet (a copy of
# docs/sciforge/cheat-sheet.md; a unit test keeps the two the same) and the Raspberry Pi 5
# case design (a copy of docs/sciforge/examples/pi5-case/pi5_case.FCStd).
# docs/ai/ (recursive): the documents for AI models the MCP server offers as resources
# (sciforge/ai/mcp_resources.py).
FILE(GLOB_RECURSE SciForge_Docs RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS docs/*.md)
FILE(GLOB_RECURSE SciForge_Examples RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    examples/*.FCStd)

SET(SciForge_ALL ${SciForge_Init} ${SciForge_Package} ${SciForge_Icons} ${SciForge_UiIcons}
    ${SciForge_Docs} ${SciForge_Examples} ${SciForge_Tests} ${SciForge_Fixtures})

ADD_CUSTOM_TARGET(SciForge ALL SOURCES ${SciForge_ALL})

fc_copy_sources(SciForge "${CMAKE_BINARY_DIR}/Mod/SciForge" ${SciForge_ALL})

INSTALL(FILES ${SciForge_Init} DESTINATION Mod/SciForge)
# DIRECTORY keeps the sub-folders (INSTALL(FILES) would put sciforge/ai/ops/core.py into
# sciforge/). The icons have their own rules below.
INSTALL(DIRECTORY sciforge DESTINATION Mod/SciForge
    FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
INSTALL(FILES ${SciForge_Icons} DESTINATION Mod/SciForge/sciforge/icons)
INSTALL(FILES ${SciForge_UiIcons} DESTINATION Mod/SciForge/sciforge/icons/ui)
INSTALL(DIRECTORY SciForgeTests DESTINATION Mod/SciForge
    FILES_MATCHING PATTERN "*.py" PATTERN "*.json" PATTERN "*.md" PATTERN "*.txt"
    PATTERN "__pycache__" EXCLUDE)
INSTALL(FILES ${SciForge_Fixtures} DESTINATION Mod/SciForge/SciForgeTests/fixtures)
INSTALL(DIRECTORY docs examples DESTINATION Mod/SciForge
    FILES_MATCHING PATTERN "*.md" PATTERN "*.FCStd")
